// --- 2. AUDIO ENGINE (Procedural Sound Design) ---
const audioCtx = typeof window !== 'undefined' ? new (window.AudioContext || window.webkitAudioContext)() : null;
const playSound = (type) => {
 if (!audioCtx || audioCtx.state === 'suspended') audioCtx?.resume();
 if (!audioCtx) return;


 const osc = audioCtx.createOscillator();
 const gain = audioCtx.createGain();
 osc.connect(gain); gain.connect(audioCtx.destination);
 const now = audioCtx.currentTime;


 if (type === 'shatter') {
   osc.type = 'triangle';
   osc.frequency.setValueAtTime(3000, now);
   osc.frequency.exponentialRampToValueAtTime(100, now + 0.3);
   gain.gain.setValueAtTime(0.2, now);
   gain.gain.exponentialRampToValueAtTime(0.01, now + 0.3);
   osc.start(now); osc.stop(now + 0.3);
 } else if (type === 'heavy') {
   osc.type = 'square';
   osc.frequency.setValueAtTime(80, now);
   osc.frequency.exponentialRampToValueAtTime(20, now + 0.5);
   gain.gain.setValueAtTime(0.3, now);
   gain.gain.exponentialRampToValueAtTime(0.01, now + 0.5);
   osc.start(now); osc.stop(now + 0.5);
 } else if (type === 'ai') {
   osc.type = 'sine';
   osc.frequency.setValueAtTime(432, now);
   osc.frequency.linearRampToValueAtTime(864, now + 1);
   gain.gain.setValueAtTime(0.1, now);
   gain.gain.linearRampToValueAtTime(0, now + 1);
   osc.start(now); osc.stop(now + 1);
 }
};


// --- 3. GEMINI API ENGINE (With Exponential Backoff) ---
const callGeminiWithRetry = async (context, query) => {
 const apiKey = "";
 const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
  const systemPrompt = `You are the AI interface of the Elementalmorphism product suite.
 Context: ${context}. The user is asking a question. Respond in 2-3 insightful, poetic, or strategic sentences.`;


 const payload = {
   contents: [{ parts: [{ text: query }] }],
   systemInstruction: { parts: [{ text: systemPrompt }] }
 };


 const delays = [1000, 2000, 4000, 8000, 16000];
 let lastError;


 for (let i = 0; i < delays.length; i++) {
   try {
     const response = await fetch(url, {
       method: 'POST',
       headers: { 'Content-Type': 'application/json' },
       body: JSON.stringify(payload)
     });
     if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
     const data = await response.json();
     return data.candidates?.[0]?.content?.parts?.[0]?.text || "The oracle is silent.";
   } catch (error) {
     lastError = error;
     if (i < delays.length - 1) await new Promise(res => setTimeout(res, delays[i]));
   }
 }
 throw lastError;
};

// --- 5. THE ADVANCED ELEMENTAL CARD ---
const EsotericCard = ({ element, title, icon, description, colors, physics, onCommune }) => {
 const cardRef = useRef(null);
 const [isHovered, setIsHovered] = useState(false);
 const [isShattered, setIsShattered] = useState(false);


 const mouseX = useMotionValue(0);
 const mouseY = useMotionValue(0);
 const springX = useSpring(mouseX, physics);
 const springY = useSpring(mouseY, physics);
  const isSingularity = element === 'singularity';
  // Inverse rotation for Singularity (Pulls towards mouse)
 const rotateX = useTransform(springY, y => {
   if (!cardRef.current) return 0;
   const h = cardRef.current.offsetHeight;
   return ((y - h / 2) / (h / 2)) * (isSingularity ? 15 : -15);
 });
 const rotateY = useTransform(springX, x => {
   if (!cardRef.current) return 0;
   const w = cardRef.current.offsetWidth;
   return ((x - w / 2) / (w / 2)) * (isSingularity ? -15 : 15);
 });


 const [fluidDrops, setFluidDrops] = useState([]);


 const handleMouseMove = (e) => {
   if (!cardRef.current) return;
   const rect = cardRef.current.getBoundingClientRect();
   const x = e.clientX - rect.left; const y = e.clientY - rect.top;
   mouseX.set(x); mouseY.set(y);


   if (element === 'relic-flux' && isHovered && Math.random() > 0.8) {
     setFluidDrops(prev => [...prev.slice(-3), { id: Date.now(), x, y, size: Math.random() * 40 + 20 }]);
   }
 };


 const handleClick = () => {
   if (element === 'prismatic-cinder') {
     playSound('shatter');
     setIsShattered(true);
     setTimeout(() => setIsShattered(false), 2000);
   } else {
     playSound('heavy');
   }
 };


 // Illusions
 const singularityShadow = useMotionTemplate`radial-gradient(400px circle at ${springX}px ${springY}px, black 0%, transparent 80%)`;
 const prismaticGlow = useMotionTemplate`radial-gradient(600px circle at ${springX}px ${springY}px, ${colors[0]}80, transparent 50%)`;


 if (isShattered) {
   return (
     <div className="relative w-full h-[400px]">
       {[...Array(6)].map((_, i) => (
         <motion.div
           key={i}
           initial={{ x: 0, y: 0, rotate: 0, opacity: 1, scale: 1 }}
           animate={{ x: (Math.random() - 0.5) * 500, y: (Math.random() - 0.5) * 500, rotate: (Math.random() - 0.5) * 360, opacity: 0, scale: 0.2 }}
           transition={{ duration: 1.5, type: "spring", damping: 12 }}
           className="absolute inset-0 bg-[#ff4b1f] border border-white/40"
           style={{
             clipPath: `polygon(${Math.random()*100}% ${Math.random()*100}%, ${Math.random()*100}% ${Math.random()*100}%, ${Math.random()*100}% ${Math.random()*100}%)`,
             boxShadow: '0 0 50px #ff4b1f'
           }}
         />
       ))}
     </div>
   );
 }


 return (
   <motion.div
     ref={cardRef} onMouseMove={handleMouseMove} onMouseEnter={() => setIsHovered(true)} onMouseLeave={() => setIsHovered(false)} onClick={handleClick}
     whileTap={{ scale: isSingularity ? 0.95 : 1.02 }}
     style={{
       rotateX, rotateY, transformPerspective: 1200,
       borderColor: isHovered ? colors[0] : 'rgba(255,255,255,0.05)',
       boxShadow: isHovered ? `0 20px 50px -10px ${colors[0]}40, inset 0 0 20px ${colors[0]}20` : '0 10px 30px -10px rgba(0,0,0,0.5)',
     }}
     className="relative p-8 h-[400px] w-full flex flex-col justify-end overflow-hidden cursor-pointer bg-white/5 border rounded-3xl transition-all duration-700"
   >
     <div className="absolute inset-0 backdrop-blur-xl z-0 pointer-events-none" />


     {/* --- LAYER 1: ILLUSIONS --- */}
     {element === 'prismatic-cinder' && (
       <motion.div style={{ background: prismaticGlow, opacity: isHovered ? 1 : 0 }} className="absolute inset-0 pointer-events-none transition-opacity duration-300 z-10 mix-blend-color-dodge filter brightness-150" />
     )}


     {element === 'relic-flux' && (
       <div className="absolute inset-0 z-10 pointer-events-none" style={{ filter: "url('#ferrofluid-goo')" }}>
         <motion.div style={{ left: springX, top: springY, x: '-50%', y: '-50%' }} className="absolute w-32 h-32 rounded-full bg-[#d4af37] opacity-80" />
         {fluidDrops.map(drop => (
           <motion.div key={drop.id} initial={{ opacity: 1, scale: 1 }} animate={{ opacity: 0, scale: 0, x: (Math.random()-0.5)*100, y: (Math.random()-0.5)*100 }} transition={{ duration: 1 }} className="absolute rounded-full bg-[#d4af37]" style={{ left: drop.x, top: drop.y, width: drop.size, height: drop.size, x: '-50%', y: '-50%' }} />
         ))}
       </div>
     )}


     {element === 'singularity' && (
       <motion.div style={{ background: singularityShadow }} className="absolute inset-0 pointer-events-none z-10 mix-blend-multiply opacity-100" />
     )}


     {/* --- LAYER 2: CONTENT --- */}
     <motion.div
       style={{ zIndex: 20 }} className="relative pointer-events-none flex flex-col h-full"
       animate={{ translateZ: isHovered ? 60 : 0 }} transition={{ type: "spring", stiffness: 100, damping: 20 }}
     >
       <div className="flex items-center gap-4 mb-3">
         <span className="text-4xl filter drop-shadow-[0_0_15px_rgba(255,255,255,0.4)]">{icon}</span>

const OracleModal = ({ element, title, color, onClose }) => {
 const [query, setQuery] = useState("");
 const [response, setResponse] = useState("");
 const [loading, setLoading] = useState(false);


 const handleSubmit = async (e) => {
   e.preventDefault();
   if (!query.trim()) return;
   setLoading(true); setResponse("");
   try {
     const res = await callGeminiWithRetry(`You are the spirit of ${title} (${element} physics).`, query);
     setResponse(res);
   } catch (err) {
     setResponse("The API connection to the elemental plane was lost.");
   } finally {
     setLoading(false);
   }
 };


 return (
   <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-xl" onClick={onClose}>
     <motion.div initial={{ scale: 0.9, y: 20 }} animate={{ scale: 1, y: 0 }} exit={{ scale: 0.9, y: 20 }} onClick={e => e.stopPropagation()} style={{ borderColor: color, boxShadow: `0 0 80px -20px ${color}` }} className="relative w-full max-w-xl p-8 rounded-3xl bg-[#050508] border backdrop-blur-2xl">
       <h3 className="text-xl font-bold text-white mb-6">Oracle of {title}</h3>
       <form onSubmit={handleSubmit} className="flex flex-col gap-4">
         <input type="text" value={query} onChange={(e) => setQuery(e.target.value)} placeholder={`Ask the ${title}...`} className="w-full px-4 py-3 text-white bg-white/5 border border-white/10 rounded-xl focus:outline-none focus:border-white/30" autoFocus />
         <button type="submit" disabled={loading} style={{ backgroundColor: `${color}40`, borderColor: color }} className="self-end px-6 py-2 text-sm text-white border rounded-full disabled:opacity-50">{loading ? "Channeling API..." : "Seek Guidance ✨"}</button>

       <button onClick={onClose} className="absolute top-6 right-6 text-white/50 hover:text-white">✕</button>

 );
