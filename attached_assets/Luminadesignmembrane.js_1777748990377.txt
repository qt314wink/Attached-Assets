const JellyCard = ({ title, description, icon: Icon, delay = '0s' }) => {
  const cardRef = useRef(null);
  const [isInteracting, setIsInteracting] = useState(false);

  // Unified physics engine for Mouse AND Touch
  const updatePhysics = (clientX, clientY) => {
    if (!cardRef.current) return;
    const card = cardRef.current;
    const rect = card.getBoundingClientRect();
    
    // Core Coordinates
    const x = clientX - rect.left;
    const y = clientY - rect.top;
    card.style.setProperty('--mouse-x', `${x}px`);
    card.style.setProperty('--mouse-y', `${y}px`);
    
    // 3D Tilt Logic
    const centerX = rect.width / 2;
    const centerY = rect.height / 2;
    const deltaX = (x - centerX) / centerX;
    const deltaY = (y - centerY) / centerY;
    
    // Reduce max rotation on mobile for better usability
    const isMobile = window.innerWidth < 768;
    const maxRotate = isMobile ? 8 : 15; 
    
    const rotateX = deltaY * -maxRotate; 
    const rotateY = deltaX * maxRotate;  
    
    card.style.transform = `perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(0.98, 0.98, 0.98)`;
  };

  const resetPhysics = () => {
    setIsInteracting(false);
    if (!cardRef.current) return;
    cardRef.current.style.transform = `perspective(1000px) rotateX(0deg) rotateY(0deg) scale3d(1, 1, 1)`;
  };

  // Mouse Handlers
  const handleMouseMove = (e) => updatePhysics(e.clientX, e.clientY);
  const handleMouseEnter = () => setIsInteracting(true);
  
  // Touch Handlers (Mobile)
  const handleTouchMove = (e) => {
    // Only track single touches
    if (e.touches.length === 1) {
      updatePhysics(e.touches[0].clientX, e.touches[0].clientY);
    }
  };
  const handleTouchStart = () => setIsInteracting(true);

  return (
    <div 
      ref={cardRef}
      onMouseMove={handleMouseMove}
      onMouseEnter={handleMouseEnter}
      onMouseLeave={resetPhysics}
      onTouchMove={handleTouchMove}
      onTouchStart={handleTouchStart}
      onTouchEnd={resetPhysics}
      className={`jelly-card p-6 md:p-8 flex flex-col items-center justify-center text-center w-full min-h-[300px] ${isInteracting ? 'is-interacting' : ''}`}
      style={{ animation: `floatUp 8s ease-in-out infinite alternate ${delay}` }}
    >
      <div className="jelly-content flex flex-col items-center">
        <div className="w-16 h-16 md:w-20 md:h-20 mb-6 rounded-[40%_60%_70%_30%/40%_50%_60%_50%] bg-gradient-to-br from-cyan-400 to-fuchsia-600 shadow-[0_0_25px_rgba(0,255,255,0.4)] flex items-center justify-center animate-[morph_5s_ease-in-out_infinite]">
          <Icon size={32} className="text-white drop-shadow-[0_0_8px_rgba(255,255,255,0.9)]" />
        </div>
        <h3 className="text-xl md:text-2xl font-bold mb-3 tracking-wide text-transparent bg-clip-text bg-gradient-to-r from-cyan-100 to-white drop-shadow-[0_0_10px_rgba(255,255,255,0.2)]">
          {title}
        </h3>
        <p className="text-cyan-50/70 leading-relaxed text-sm md:text-base">
          {description}
        </p>
      </div>
    </div>
  );
};
