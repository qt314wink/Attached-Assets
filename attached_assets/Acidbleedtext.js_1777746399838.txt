import React, { useState, useEffect, useRef, useCallback } from 'react';

// =====================================================================
// 1. SENSORY & HAPTIC ENGINE
// =====================================================================
const useAudioSynth = () => {
  const audioCtx = useRef(null);

  const initAudio = () => {
    if (!audioCtx.current) {
      audioCtx.current = new (window.AudioContext || window.webkitAudioContext)();
    }
  };

  const playTone = (freq, type, duration, vol = 0.1) => {
    if (!audioCtx.current) return;
    const osc = audioCtx.current.createOscillator();
    const gain = audioCtx.current.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, audioCtx.current.currentTime);
    gain.gain.setValueAtTime(vol, audioCtx.current.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.current.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.current.destination);
    osc.start();
    osc.stop(audioCtx.current.currentTime + duration);
  };

  const playBloop = () => { initAudio(); playTone(300, 'sine', 0.2, 0.15); setTimeout(() => playTone(400, 'sine', 0.1, 0.1), 100); };
  const playSizzle = () => { initAudio(); playTone(800, 'sawtooth', 0.05, 0.02); };
  const playPop = () => { initAudio(); playTone(500, 'sine', 0.1, 0.1); };

  return { playBloop, playSizzle, playPop, initAudio };
};

const triggerHaptic = (pattern) => {
  if (typeof window !== 'undefined' && window.navigator && window.navigator.vibrate) {
    window.navigator.vibrate(pattern);
  }
};

// --- A. Acid Bubble / Paint Bleed Button ---
// Uses the SVG #goo filter to make internal rotating divs "melt" into each other.
const AcidBubbleButton = ({ text, colorClass = "bg-pink-400" }) => {
  const { playBloop } = useAudioSynth();
  const [isHovered, setIsHovered] = useState(false);

  const handleInteraction = () => {
    playBloop();
    triggerHaptic([15, 30, 20]);
  };

  return (
    <div className="relative group cursor-pointer" 
         onMouseEnter={() => setIsHovered(true)}
         onMouseLeave={() => setIsHovered(false)}
         onClick={handleInteraction}
         style={{ filter: "url(#goo)" }}> {/* The Magic Liquid Filter */}
      
      {/* The main button body */}
      <div className={`relative z-20 px-8 py-4 rounded-[2rem] ${colorClass} text-white font-black text-xl transition-transform duration-300 ${isHovered ? 'scale-105' : 'scale-100'}`}>
        {text}
      </div>

      {/* Orbiting Gooey Blobs */}
      <div className={`absolute top-0 left-4 w-8 h-8 rounded-full ${colorClass} transition-all duration-700 ease-in-out ${isHovered ? '-translate-y-8 translate-x-4 scale-150' : 'translate-y-0 scale-50'}`} />
      <div className={`absolute bottom-0 right-4 w-6 h-6 rounded-full ${colorClass} transition-all duration-500 ease-in-out delay-75 ${isHovered ? 'translate-y-6 -translate-x-6 scale-125' : 'translate-y-0 scale-50'}`} />
      <div className={`absolute top-1/2 -left-2 w-10 h-10 rounded-full ${colorClass} transition-all duration-1000 ease-in-out ${isHovered ? '-translate-x-8 -translate-y-4 scale-110' : 'translate-x-0 scale-0'}`} />
    </div>
  );
};

// --- B. Liquid Splatter / Paint Pour Engine ---
// Particles spawn inside a gooey container, creating liquid surface tension merges.
const LiquidSplatterContainer = () => {
  const [droplets, setDroplets] = useState([]);
  const containerRef = useRef(null);
  const { playPop, playSizzle } = useAudioSynth();

  const handlePour = (e) => {
    playSizzle();
    triggerHaptic(10);
    const rect = containerRef.current.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;

    const newDroplets = Array.from({ length: 8 }).map((_, i) => ({
      id: Date.now() + i,
      x: x + (Math.random() - 0.5) * 40,
      y: y + (Math.random() - 0.5) * 40,
      size: Math.random() * 30 + 10,
      color: ['bg-purple-400', 'bg-pink-400', 'bg-mint-400', 'bg-yellow-300'][Math.floor(Math.random() * 4)]
    }));

    setDroplets(prev => [...prev, ...newDroplets]);

    // Melt away
    setTimeout(() => {
      setDroplets(prev => prev.filter(p => !newDroplets.find(np => np.id === p.id)));
    }, 2500);
  };

  return (
    <div 
      ref={containerRef}
      onMouseMove={(e) => { if (e.buttons === 1) handlePour(e); }}
      onClick={(e) => { handlePour(e); playPop(); triggerHaptic([20, 40, 20]); }}
      className="relative w-full h-64 bg-white/30 rounded-[3rem] border border-white/50 shadow-inner overflow-hidden cursor-crosshair"
      style={{ filter: "url(#goo)" }}
    >
      <div className="absolute inset-0 flex items-center justify-center opacity-50 pointer-events-none">
        <span className="font-black text-2xl text-deep-purple uppercase tracking-widest">Click & Drag to Pour Paint</span>
      </div>

      {droplets.map(drop => (
        <div
          key={drop.id}
          className={`absolute rounded-full ${drop.color} animate-melt-away`}
          style={{
            left: drop.x - drop.size/2,
            top: drop.y - drop.size/2,
            width: drop.size,
            height: drop.size,
          }}
        />
      ))}
    </div>
  );
};

// --- D. Liquid Surface Tension Slider ---
// Track and thumb merge together physically.
const GooeySlider = () => {
  const [val, setVal] = useState(50);
  const { playSizzle } = useAudioSynth();

  const handleInput = (e) => {
    setVal(e.target.value);
    if (e.target.value % 10 === 0) { playSizzle(); triggerHaptic(5); }
  };

  return (
    <div className="relative w-full h-24 flex items-center justify-center" style={{ filter: "url(#goo)" }}>
      {/* Track */}
      <div className="absolute w-full h-4 bg-purple-300 rounded-full" />
      
      {/* Actual Input */}
      <input 
        type="range" min="0" max="100" value={val} onChange={handleInput}
        className="absolute w-full h-full opacity-0 z-20 cursor-grab active:cursor-grabbing"
      />
      
      {/* Gooey Thumb */}
      <div 
        className="absolute w-12 h-12 bg-pink-400 rounded-full z-10 transition-transform duration-100 ease-out flex items-center justify-center shadow-inner"
        style={{ left: `calc(${val}% - 24px)` }}
      >
        <div className="w-4 h-4 bg-white/60 rounded-full blur-[1px]"></div>
      </div>
      
      {/* Connecting Droplets (Surface Tension) */}
      <div className="absolute w-6 h-6 bg-purple-300 rounded-full transition-all duration-300" style={{ left: `calc(${val}% - 12px)`, opacity: val < 10 || val > 90 ? 0 : 1 }} />
    </div>
  );
};
