import React, { useState, useEffect, useRef } from 'react';

// =====================================================================
// 1. SENSORY & HAPTIC ENGINE
// =====================================================================
const useAudioSynth = () => {
  const audioCtx = useRef(null);

  const initAudio = () => {
    if (!audioCtx.current) {
      audioCtx.current = new (window.AudioContext || window.webkitAudioContext)();
    }
  };

  const playTone = (freq, type, duration, vol = 0.1) => {
    if (!audioCtx.current) return;
    const osc = audioCtx.current.createOscillator();
    const gain = audioCtx.current.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, audioCtx.current.currentTime);
    gain.gain.setValueAtTime(vol, audioCtx.current.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.current.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.current.destination);
    osc.start();
    osc.stop(audioCtx.current.currentTime + duration);
  };

  const playBloop = () => { initAudio(); playTone(350, 'sine', 0.2, 0.1); setTimeout(() => playTone(450, 'sine', 0.1, 0.1), 80); };
  const playSlurp = () => { initAudio(); playTone(200, 'sine', 0.4, 0.15); };
  const playPop = () => { initAudio(); playTone(600, 'sine', 0.05, 0.15); };
  const playDrip = () => { initAudio(); playTone(1200, 'sine', 0.1, 0.05); };
  const playSplat = () => { initAudio(); playTone(150, 'sawtooth', 0.1, 0.1); };

  return { playBloop, playSlurp, playPop, playDrip, playSplat, initAudio };
};

const triggerHaptic = (pattern) => {
  if (typeof window !== 'undefined' && window.navigator && window.navigator.vibrate) {
    window.navigator.vibrate(pattern);
  }
};

// --- A. Gooey Acid Bubble FAB (Peekaboo Menu) ---
const AcidBubbleMenu = () => {
  const [isOpen, setIsOpen] = useState(false);
  const { playBloop, playDrip } = useAudioSynth();
  const actions = ['🧪', '🦠', '🔮'];

  const toggle = () => {
    if (!isOpen) playBloop(); else playDrip();
    triggerHaptic([15, 30, 15]);
    setIsOpen(!isOpen);
  };

  return (
    <div className="relative w-24 h-64 flex flex-col items-center justify-end">
      <div className="absolute inset-0 pointer-events-none" style={{ filter: "url(#goo-viscous)" }}>
        {actions.map((_, i) => (
          <div 
            key={i}
            className="absolute bottom-4 left-1/2 -translate-x-1/2 w-16 h-16 bg-pink-400 rounded-full transition-all duration-500 ease-spring"
            style={{ 
              transform: `translate(-50%, ${isOpen ? -(i + 1) * 70 : 0}px) scale(${isOpen ? 1 : 0.4})`,
              transitionDelay: `${isOpen ? i * 50 : (2 - i) * 50}ms`,
              opacity: isOpen ? 1 : 0
            }}
          />
        ))}
        <div className="absolute bottom-4 left-1/2 -translate-x-1/2 w-20 h-20 bg-pink-500 rounded-full shadow-lg border-4 border-white/20" />
      </div>

      {actions.map((icon, i) => (
        <button
          key={i}
          className={`absolute left-1/2 -translate-x-1/2 w-16 h-16 flex items-center justify-center text-2xl z-10 transition-all duration-500 ease-spring ${isOpen ? 'opacity-100 cursor-pointer scale-100 hover:scale-110' : 'opacity-0 pointer-events-none scale-50'}`}
          style={{ 
            transform: `translate(-50%, ${isOpen ? -(i + 1) * 70 : 0}px)`,
            transitionDelay: `${isOpen ? i * 50 : (2 - i) * 50}ms`
          }}
          onClick={() => triggerHaptic(10)}
        >
          {icon}
        </button>
      ))}

      <button onClick={toggle} className="absolute bottom-4 left-1/2 -translate-x-1/2 w-20 h-20 z-20 flex items-center justify-center text-white text-4xl font-black active:scale-90 transition-transform">
        <div className={`transition-transform duration-500 ${isOpen ? 'rotate-45' : 'rotate-0'}`}>+</div>
      </button>
    </div>
  );
};

// --- B. High-Tension Liquid Slider (Rubber-Band Bridge) ---
const GooeySlider = ({ value, onChange }) => {
  const [trail, setTrail] = useState(value);
  const { playPop } = useAudioSynth();

  useEffect(() => {
    const timeout = setTimeout(() => setTrail(value), 120);
    return () => clearTimeout(timeout);
  }, [value]);

  const handleInput = (v) => {
    if (Math.abs(v - value) > 10) playPop();
    onChange(v);
  };

  return (
    <div className="relative w-full h-32 flex items-center" style={{ filter: "url(#goo-viscous)" }}>
      {/* Background Track */}
      <div className="absolute w-full h-6 bg-purple-100 rounded-full" />
      
      {/* Liquid Bridge (The Stretch) */}
      <div 
        className="absolute h-10 bg-pink-500 rounded-full transition-all duration-75 opacity-90"
        style={{ 
          left: `calc(${Math.min(value, trail)}% - 8px)`, 
          width: `calc(${Math.abs(value - trail)}% + 32px)`,
        }}
      />
      <input 
        type="range" min="0" max="100" value={value} 
        onChange={(e) => handleInput(e.target.value)}
        className="absolute w-full h-full opacity-0 z-50 cursor-pointer"
      />
      {/* The Thumb */}
      <div className="absolute w-16 h-16 bg-pink-500 rounded-full shadow-lg border-4 border-white z-40 pointer-events-none" 
           style={{ left: `calc(${value}% - 32px)` }}>
        <div className="w-4 h-4 bg-white/60 rounded-full blur-[1px] m-auto mt-3" />
      </div>
    </div>
  );
};
