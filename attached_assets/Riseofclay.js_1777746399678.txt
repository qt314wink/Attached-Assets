import React, { useState, useEffect, useRef, useCallback } from 'react';

const useAudioSynth = () => {
  const audioCtx = useRef(null);

  const initAudio = () => {
    if (!audioCtx.current) {
      audioCtx.current = new (window.AudioContext || window.webkitAudioContext)();
    }
  };

  const playTone = (freq, type, duration, vol = 0.1) => {
    if (!audioCtx.current) return;
    const osc = audioCtx.current.createOscillator();
    const gain = audioCtx.current.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, audioCtx.current.currentTime);
    gain.gain.setValueAtTime(vol, audioCtx.current.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.current.currentTime + duration);
    osc.connect(gain);
    gain.connect(audioCtx.current.destination);
    osc.start();
    osc.stop(audioCtx.current.currentTime + duration);
  };

  // Rule of Numbness: Varying strengths for different actions
  const playPop = () => { initAudio(); playTone(450, 'sine', 0.1, 0.15); }; // Medium action
  const playTick = () => { initAudio(); playTone(850, 'triangle', 0.05, 0.05); }; // Frequent action
  const playSuccess = () => { // Rare action
    initAudio();
    playTone(440, 'sine', 0.1, 0.1);
    setTimeout(() => playTone(554, 'sine', 0.1, 0.15), 100);
    setTimeout(() => playTone(659, 'sine', 0.2, 0.2), 200);
    setTimeout(() => playTone(880, 'sine', 0.4, 0.25), 400);
  };

  return { playPop, playTick, playSuccess, initAudio };
};

const triggerHaptic = (pattern) => {
  if (typeof window !== 'undefined' && window.navigator && window.navigator.vibrate) {
    window.navigator.vibrate(pattern);
  }
};

const ScrambleText = ({ text }) => {
  const [displayText, setDisplayText] = useState(text);
  const [isHovered, setIsHovered] = useState(false);
  const chars = '✦✧★☆✶✵❂✡✢✣✤✥';
  
  useEffect(() => {
    let interval;
    if (isHovered) {
      let iteration = 0;
      interval = setInterval(() => {
        setDisplayText((prev) =>
          text.split('').map((letter, index) => {
            if (index < iteration) return text[index];
            return chars[Math.floor(Math.random() * chars.length)];
          }).join('')
        );
        if (iteration >= text.length) clearInterval(interval);
        iteration += 1 / 3;
      }, 30);
    } else {
      setDisplayText(text);
    }
    return () => clearInterval(interval);
  }, [isHovered, text]);

  return (
    <span 
      onMouseEnter={() => setIsHovered(true)} 
      onMouseLeave={() => setIsHovered(false)}
      className="inline-block transition-transform duration-300 hover:scale-105 cursor-default"
    >
      {displayText}
    </span>
  );
};

const ClayButton = ({ children, onClick, colorClass = "clay-bg-purple", className = "", sound = "pop", disabled=false }) => {
  const { playPop, playTick } = useAudioSynth();
  
  const handlePress = (e) => {
    if (disabled) return;
    // Haptic Mapping
    if (sound === "pop") { playPop(); triggerHaptic(15); }
    if (sound === "tick") { playTick(); triggerHaptic(5); } // Rule of numbness: lighter touch
    if (onClick) onClick(e);
  };

  return (
    <button 
      onClick={handlePress}
      disabled={disabled}
      className={`clay-base clay-squish ${colorClass} ${className} ${disabled ? 'opacity-50 cursor-not-allowed' : ''} font-bold text-deep-purple px-6 py-3 select-none flex items-center justify-center gap-2`}
    >
      {children}
    </button>
  );
};

const PsychedelicGradientDemo = () => {
  const containerRef = useRef(null);
  const [mousePos, setMousePos] = useState({ x: 50, y: 50 });

  const handleMouseMove = (e) => {
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const x = ((e.clientX - rect.left) / rect.width) * 100;
    const y = ((e.clientY - rect.top) / rect.height) * 100;
    setMousePos({ x, y });
  };

  return (
    <div 
      ref={containerRef}
      onMouseMove={handleMouseMove}
      className="relative w-full h-64 rounded-[3rem] overflow-hidden border-4 border-white/60 shadow-inner cursor-crosshair"
    >
      <div 
        className="absolute inset-0 transition-transform duration-75 ease-out"
