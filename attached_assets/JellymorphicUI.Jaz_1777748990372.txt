const JellyCard = ({ title, description, icon: Icon, delay = '0s' }) => {
  const cardRef = useRef(null);
  const [isHovered, setIsHovered] = useState(false);

  const handleMouseMove = (e) => {
    if (!cardRef.current) return;
    const card = cardRef.current;
    const rect = card.getBoundingClientRect();
    
    // 1. Calculate Glow Position
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    card.style.setProperty('--mouse-x', `${x}px`);
    card.style.setProperty('--mouse-y', `${y}px`);
    
    // 2. Calculate 3D Tilt Physics
    const centerX = rect.width / 2;
    const centerY = rect.height / 2;
    
    // Normalize coordinates (-1 to 1)
    const deltaX = (x - centerX) / centerX;
    const deltaY = (y - centerY) / centerY;
    
    // Max rotation in degrees
    const maxRotate = 15;
    const rotateX = deltaY * -maxRotate; 
    const rotateY = deltaX * maxRotate;  
    
    // Apply Transform directly to DOM for 60fps performance (bypassing React State)
    card.style.transform = `perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(0.98, 0.98, 0.98)`;
  };

  const handleMouseEnter = () => setIsHovered(true);

  const handleMouseLeave = () => {
    setIsHovered(false);
    if (!cardRef.current) return;
    const card = cardRef.current;
    // Reset transform. The CSS transition handles the smooth "snap back"
    card.style.transform = `perspective(1000px) rotateX(0deg) rotateY(0deg) scale3d(1, 1, 1)`;
  };

  return (
    <div 
      ref={cardRef}
      onMouseMove={handleMouseMove}
      onMouseEnter={handleMouseEnter}
      onMouseLeave={handleMouseLeave}
      className={`jelly-card p-8 min-h-[320px] flex flex-col items-center justify-center text-center ${isHovered ? 'is-hovered' : ''}`}
      style={{ animation: `floatOrganism 6s ease-in-out infinite ${delay}` }}
    >
      <div className="jelly-content flex flex-col items-center">
        <div className="w-20 h-20 mb-6 rounded-[40%_60%_70%_30%/40%_50%_60%_50%] bg-gradient-to-br from-cyan-400 to-fuchsia-500 shadow-[0_0_20px_rgba(0,255,255,0.5)] flex items-center justify-center animate-[morph_5s_ease-in-out_infinite]">

        </div>

          {title}
        </h3>
        <p className="text-cyan-50/70 leading-relaxed text-sm">
          {description}
        </p>
      </div>
    </div>
  );
};
